public class FloorProfile : Object {
 public class RoomPool { public RoomKind kind; public RoomInstance[] prefabs; }
 public RoomPool[] roomPools;
 public int RoomCount,EliteRoomCount,TreasureRoomCount,ShopRoomCount,GenerationAttempts=128;
 public bool IsBossStage,KeepNonCombatRoomsAwayFromStart; public string DisplayName;
 public Vector2 GridCellSize => new Vector2(48,32);
public List<RoomInstance> GetRoomPrefabs(RoomKind kind)
    {
        List<RoomInstance> valid = new List<RoomInstance>();
        if (roomPools == null) return valid;
        foreach (RoomPool pool in roomPools)
        {
            if (pool == null || pool.kind != kind || pool.prefabs == null) continue;
            foreach (RoomInstance prefab in pool.prefabs)
            {
                if (prefab != null && prefab.Kind == kind && !valid.Contains(prefab)) valid.Add(prefab);
            }
        }
        return valid;
    }
}


/// <summary>방이 이웃과 이어지는 네 방향.</summary>
public enum RoomDirection
{
    North,
    East,
    South,
    West
}

/// <summary>방의 역할. 층 생성기가 종류별로 프리팹 풀을 골라 배치한다.</summary>
public enum RoomKind
{
    Start,
    Normal,
    Elite,
    Treasure,
    Shop,
    Boss
}

public static class RoomDirectionUtility
{
    public static readonly RoomDirection[] All =
    {
        RoomDirection.North,
        RoomDirection.East,
        RoomDirection.South,
        RoomDirection.West
    };

    public static RoomDirection Opposite(this RoomDirection direction)
    {
        switch (direction)
        {
            case RoomDirection.North: return RoomDirection.South;
            case RoomDirection.South: return RoomDirection.North;
            case RoomDirection.East: return RoomDirection.West;
            default: return RoomDirection.East;
        }
    }

    /// <summary>그리드 좌표 기준 이웃 칸 오프셋.</summary>
    public static Vector2Int ToOffset(this RoomDirection direction)
    {
        switch (direction)
        {
            case RoomDirection.North: return new Vector2Int(0, 1);
            case RoomDirection.South: return new Vector2Int(0, -1);
            case RoomDirection.East: return new Vector2Int(1, 0);
            default: return new Vector2Int(-1, 0);
        }
    }
}



/// <summary>생성된 층 한 장의 결과물.</summary>
public static class StageRulesHarness {
 static void Check(bool ok,string label) { if (!ok) throw new Exception(label); }
 static Dictionary<Vector2Int,int> Distances(List<Vector2Int> cells) {
  var valid=new HashSet<Vector2Int>(cells); var dist=new Dictionary<Vector2Int,int> {{Vector2Int.zero,0}}; var q=new Queue<Vector2Int>();q.Enqueue(Vector2Int.zero);
  while(q.Count>0) { var c=q.Dequeue(); foreach(var dir in RoomDirectionUtility.All) { var n=c+dir.ToOffset(); if(valid.Contains(n)&&!dist.ContainsKey(n)) {dist[n]=dist[c]+1;q.Enqueue(n);} } } return dist;
 }
 static FloorProfile[] Profiles() => new FloorProfile[]{};
 public static void Run() {
  foreach(var p in Profiles()) {
   for(int seed=0;seed<1000;seed++) {
    Random.InitState(seed);
    Check(FloorGenerator.TryPlan(p,out var cells,out var kinds,out var prefabs),p.DisplayName+" failed plan seed "+seed);
    var distances=Distances(cells);
    Check(cells.Count==p.RoomCount&&distances.Count==p.RoomCount,"room count/connectivity");
    Check(kinds.Values.Count(k=>k==RoomKind.Start)==1&&kinds.Values.Count(k=>k==RoomKind.Boss)==(p.IsBossStage?1:0),"boss only in separate boss stage");
    Check(kinds.Values.Count(k=>k==RoomKind.Elite)==p.EliteRoomCount&&kinds.Values.Count(k=>k==RoomKind.Treasure)==p.TreasureRoomCount&&kinds.Values.Count(k=>k==RoomKind.Shop)==p.ShopRoomCount,"special counts preserved");
    foreach(var c in cells) {
     var k=kinds[c]; Check(prefabs[c].Kind==k,"prefab kind matches role");
     if(k==RoomKind.Elite) Check(distances[c]>=2,"elite start distance");
     if(k==RoomKind.Boss) Check(distances[c]==distances.Values.Max(),"farthest boss");
     if(p.KeepNonCombatRoomsAwayFromStart&&distances[c]==1) Check(k==RoomKind.Normal,"first stage neighbour normal");
     foreach(var d in RoomDirectionUtility.All) if(prefabs.TryGetValue(c+d.ToOffset(),out var n)) Check(prefabs[c].DesignId!=n.DesignId,"adjacent design repetition");
    }
   }
   Random.InitState(73); Object.RoomSpawns=0;Object.ExitSpawns=0;
   var floor=FloorGenerator.Generate(p,new RoomRuntimeContext(),null);
   Check(floor!=null&&floor.Exit!=null&&Object.ExitSpawns==1&&Object.RoomSpawns==p.RoomCount,"exactly one instantiated exit");
   Check(floor.ExitRoom!=null&&floor.ExitRoom.IsCombatRoom,"exit belongs to combat room");
   Check(!floor.IsCombatCleared,"no premature clear");
   foreach(var room in floor.Rooms) if(room.IsCombatRoom) room.IsCleared=true;
   Check(floor.IsCombatCleared,"all combat clear opens exit without requiring rewards");
   floor.ExitRoom.IsCleared=false;
   Check(!floor.IsCombatCleared,"uncleared exit room blocks exit");
   Console.WriteLine("PASS: "+p.DisplayName+" 1000 layouts, all distances/designs/counts and one exit");
  }
  var invalid=Profiles().First(p=>!p.IsBossStage); invalid.RoomCount=3; Object.RoomSpawns=0;Object.ExitSpawns=0;Random.InitState(1);
  Check(FloorGenerator.Generate(invalid,new RoomRuntimeContext(),null)==null&&Object.RoomSpawns==0&&Object.ExitSpawns==0,"impossible counts create no partial floor");
  invalid=Profiles()[0];foreach(var pool in invalid.roomPools) foreach(var room in pool.prefabs) room.DesignId="same";
  Check(!FloorGenerator.TryPlan(invalid,out _,out _,out _),"impossible shared design must fail, not bypass rules");
  invalid=Profiles()[0];foreach(var pool in invalid.roomPools) foreach(var room in pool.prefabs) room.Mask=0;
  Check(!FloorGenerator.TryPlan(invalid,out _,out _,out _),"missing room doors must fail");
  Check(FloorGenerator.Generate(Profiles()[0],new RoomRuntimeContext {ExitPrefab=null},null)==null,"missing exit prefab must fail");
  Console.WriteLine("PASS: invalid counts, impossible designs, missing doors/exit rejected without partial maps.");
 }
